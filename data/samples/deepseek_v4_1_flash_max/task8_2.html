import express from "express";
import { verifySignature } from "@patchbay/verify";

const app = express();

// Keep the raw body — JSON parsing breaks the signature.
app.post("/hooks/payments", express.raw({ type: "*/*" }), async (req, res) => {
  const ok = verifySignature({
    scheme:  "stripe",
    secret:  process.env.PATCHBAY_SIGNING_SECRET,
    payload: req.body,
    header:  req.get("stripe-signature"),
  });

  if (!ok) return res.status(400).send("invalid signature");

  const event = JSON.parse(req.body);
  if (event.type === "payment_intent.succeeded") {
    await fulfilOrder(event.data.object.id, event.data.object.amount);
  }

  res.status(200).json({ received: true });
});

import json, os
from fastapi import FastAPI, Request, HTTPException
from patchbay import verify_signature

app = FastAPI()

@app.post("/hooks/payments")
async def stripe_hook(request: Request):
    body = await request.body()

    if not verify_signature(
        scheme="stripe",
        secret=os.environ["PATCHBAY_SIGNING_SECRET"],
        payload=body,
        header=request.headers.get("stripe-signature"),
    ):
        raise HTTPException(status_code=400, detail="invalid signature")

    event = json.loads(body)
    if event["type"] == "payment_intent.succeeded":
        await fulfil_order(event["data"]["object"]["id"])

    return {"received": True}

package main

import (
	"io"
	"net/http"
	"os"

	"github.com/patchbay/verify-go"
)

func main() {
	http.HandleFunc("/hooks/payments", func(w http.ResponseWriter, r *http.Request) {
		body, _ := io.ReadAll(r.Body)

		if err := verify.Check(verify.Options{
			Scheme:  "stripe",
			Secret:  os.Getenv("PATCHBAY_SIGNING_SECRET"),
			Payload: body,
			Header:  r.Header.Get("Stripe-Signature"),
		}); err != nil {
			http.Error(w, "invalid signature", http.StatusBadRequest)
			return
		}

		w.WriteHeader(http.StatusOK)
		w.Write([]byte(`{"received":true}`))
	})

	http.ListenAndServe(":3000", nil)
}

FAQ

Questions from the queue.

Still unsure? Write to beta@patchbay.dev and a human on the platform team replies, usually within a day.

What exactly does "temporary endpoint" mean?

Every endpoint gets a unique URL such as in.patchbay.dev/e/9f2c1a that accepts inbound HTTP traffic from any provider. It stays live for the lifetime you choose — 1 hour, 7 days, or until you delete it — and it holds every request it received, including the raw body, headers and signature, so nothing is lost when the endpoint expires.

Do I need to open a firewall port or run a tunnel?

No. Patchbay receives the traffic on our infrastructure, so your local machine and your staging network stay closed to the internet. Point the provider at the Patchbay URL, watch the events arrive, and when you are ready to wire things up for real, forward captured events to your own handler with the replay controls.

Which signing methods are supported in the beta?

The private beta covers HTTP webhooks and the common signature schemes: Stripe (t= plus v1= HMAC-SHA256), GitHub's X-Hub-Signature-256, generic HMAC-SHA256 and SHA-1 with a custom header, plus the Shopify, Twilio and Slack header formats. Async transports — AMQP, Kafka, SQS — and gRPC/WebSocket streams are planned for Q3 and Q4 2025.

Can I replay an event more than once, or edit it first?

Both. Replays are unlimited during beta. You can re-send the original bytes untouched, edit the JSON in the browser before sending, or apply a failure profile that forces a 400, 429 or 500 response with optional added latency. Every replay is recorded as a new attempt against the original event, so the comparison stays honest.

How does sharing work across engineering and support?

A debugging session is a link that carries the event, its headers, the signature check result and the comment thread. Anyone in your workspace can open it without installing anything, and support staff can be given read-only access so they see the same payload engineering sees. Sessions are retained for as long as the underlying event is.

What happens to my data when the beta ends?

You keep it. Workspaces migrated to a paid plan keep their event history and session comments, and you can export raw payloads as JSON or NDJSON at any time. If you decide not to continue, we delete your workspace data 30 days after you request it and send a written confirmation.

Is there a cost during the private beta?

No. The beta is free for the whole cohort, including unlimited endpoints and 30 days of event retention, capped at 5 million captured events per workspace. When we launch publicly, beta teams get their pricing locked for 12 months and 30 days of notice before any change takes effect.
